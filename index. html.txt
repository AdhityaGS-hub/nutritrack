        // Web Bluetooth API Integration
        async function requestBluetoothDevice() {
            if (!navigator.bluetooth) {
                showToast('Web Bluetooth API is not supported in this browser.', 'error');
                return;
            }

            try {
                showToast('Searching for BLE devices...', 'info');
                
                // Requesting Bluetooth GATT Heart Rate Device
                const device = await navigator.bluetooth.requestDevice({
                    filters: [{ services: ['heart_rate'] }],
                    optionalServices: ['battery_service']
                });

                state.ble.device = device;
                updateBLEUIStatus('Connecting...', 'amber');

                device.addEventListener('gattserverdisconnected', onBLEDisconnected);

                const server = await device.gatt.connect();
                state.ble.server = server;
                state.ble.connected = true;

                showToast(`Connected to ${device.name || 'GATT Device'}`, 'success');
                updateBLEUIStatus('Connected', 'emerald');
                document.getElementById('ble-device-name').innerText = device.name || 'BLE Health Sensor';
                document.getElementById('ble-device-id').innerText = device.id.slice(0, 12);

                // Start Heart Rate Service Telemetry
                const hrService = await server.getPrimaryService('heart_rate');
                const hrChar = await hrService.getCharacteristic('heart_rate_measurement');
                await hrChar.startNotifications();
                hrChar.addEventListener('characteristicvaluechanged', handleHeartRateNotification);

                // Battery Service
                try {
                    const battService = await server.getPrimaryService('battery_service');
                    const battChar = await battService.getCharacteristic('battery_level');
                    const battVal = await battChar.readValue();
                    state.ble.battery = battVal.getUint8(0);
                    updateBatteryUI(state.ble.battery);
                } catch (e) {
                    console.log('Battery service not exposed by device');
                }

            } catch (err) {
                console.error('BLE Pair error:', err);
                showToast('Bluetooth pairing canceled or failed.', 'error');
                updateBLEUIStatus('Disconnected', 'slate');
            }
        }

        function handleHeartRateNotification(event) {
            const value = event.target.value;
            const flags = value.getUint8(0);
            let rate = 0;
            if (flags & 0x01) {
                rate = value.getUint16(1, true);
            } else {
                rate = value.getUint8(1);
            }
            processNewHeartRate(rate);
        }

        function processNewHeartRate(rate) {
            state.ble.heartRate = rate;

            // Live calorie burn accumulation estimation formula based on heart rate
            const calBurnPerMin = ((rate * 0.2017) - (state.user.weight * 0.09036) + (state.user.age * 0.074) - 20.4022) / 4.184;
            const rateKcalMin = Math.max(0.1, calBurnPerMin).toFixed(1);

            state.ble.activeBurn += Math.round(calBurnPerMin / 12); // updated per 5s interval equivalent

            // Update UI
            document.getElementById('ble-live-bpm').innerText = rate;
            document.getElementById('dash-hr-display').innerText = `${rate} BPM`;
            document.getElementById('ble-header-hr').innerText = `${rate} BPM`;
            document.getElementById('ble-cal-rate').innerText = `${rateKcalMin} kcal/min`;
            document.getElementById('ble-session-burn').innerText = `${state.ble.activeBurn} kcal`;
            document.getElementById('dash-active-burn').innerText = state.ble.activeBurn;
            document.getElementById('ring-burn-cal').innerText = state.ble.activeBurn;
            document.getElementById('dash-ble-burn-note').innerText = `Live BLE Rate: ~${rate} bpm`;

            // Chart update
            state.ble.history.push(rate);
            if (state.ble.history.length > 15) state.ble.history.shift();
            if (bleChart) {
                bleChart.data.datasets[0].data = state.ble.history;
                bleChart.update();
            }

            renderCalorieRing();
        }

        function calculateBMI() {
            const hM = state.user.height / 100;
            const bmi = (state.user.weight / (hM * hM)).toFixed(1);
            let category = 'Normal';
            if (bmi < 18.5) category = 'Underweight';
            else if (bmi >= 25 && bmi < 29.9) category = 'Overweight';
            else if (bmi >= 30) category = 'Obese';

            return `${bmi} (${category})`;
        }

        function saveLocalStorage() {
            localStorage.setItem('nutritrack_data', JSON.stringify({
                user: state.user,
                goals: state.goals,

                waterLogs: state.waterLogs

        function loadLocalStorage() {
            const saved = localStorage.getItem('nutritrack_data');
            if (saved) {
                try {
                    const parsed = JSON.parse(saved);
                    if (parsed.user) state.user = parsed.user;
                    if (parsed.goals) state.goals = parsed.goals;
                    if (parsed.meals) state.meals = parsed.meals;
                    if (parsed.waterLogs) state.waterLogs = parsed.waterLogs;
                } catch (e) {
                    console.error('Error reading localStorage', e);
                }
            }
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');

            const colors = {
                success: 'border-emerald-500 bg-emerald-950/90 text-emerald-200',
                error: 'border-rose-500 bg-rose-950/90 text-rose-200',
                warning: 'border-amber-500 bg-amber-950/90 text-amber-200',
                info: 'border-cyan-500 bg-cyan-950/90 text-cyan-200'
            };

            toast.className = `p-3.5 rounded-xl border backdrop-blur-md shadow-lg text-xs font-semibold flex items-center gap-2 transition-all duration-300 transform translate-y-2 opacity-0 pointer-events-auto ${colors[type] || colors.info}`;
            toast.innerHTML = `<span>${message}</span>`;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
